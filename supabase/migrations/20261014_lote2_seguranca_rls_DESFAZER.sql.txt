create or replace function public.user_can_access_company(company_uuid uuid)
returns boolean language sql stable security definer set search_path to 'public'
as $function$
  select coalesce(public.is_admin(), exists (
    select 1 from public.companies c
    where c.id = company_uuid and (c.user_id = auth.uid() or c.responsible_user_id = auth.uid())
  ), false);
$function$;
create or replace function public.user_owns_company(company_uuid uuid)
returns boolean language sql stable security definer set search_path to 'public'
as $function$
  select coalesce(public.is_admin(), exists (
    select 1 from public.companies c where c.id = company_uuid and c.user_id = auth.uid()
  ), false);
$function$;

create or replace function public.portal_has_edit(screen text)
returns boolean language sql stable security definer set search_path to 'public'
as $function$
  select coalesce(public.is_admin(), (
    select coalesce(p.is_active, true)
      and (p.portal_permissions is null or coalesce(p.portal_permissions ->> screen, '') = 'edit')
    from public.profiles p where p.id = auth.uid()
  ), false);
$function$;
